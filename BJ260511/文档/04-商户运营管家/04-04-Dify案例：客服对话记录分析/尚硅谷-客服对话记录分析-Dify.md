### 4.将ROWS处理为LIST

#### 1.配置

![image-20260130104509659](images/image-20260130104509659.png)

![image-20260130104526264](images/image-20260130104526264.png)

代码内容如下

```python
import re
import json

# 定义入口函数main，接收字符串类型参数text
def main(text: str):
    # 将文本里字面量字符串"\n"替换为真实换行符，字面量"\t"替换为真实制表符，兼容转义后的文本数据
    text = text.replace("\\n", "\n").replace("\\t", "\t")
    # 预编译正则表达式，匹配以Row+空格+数字+冒号开头的行，捕获冒号后全部内容
    row_re = re.compile(r"^Row\s+\d+:\s*(.*)$")
    # 定义变量header，用于存储表头字段，初始为空
    header = None
    # 定义空列表out，用来存放最终解析完成的字典数据
    out = []
    # 将处理完换行的文本按行分割，循环遍历每一行文本
    for line in text.splitlines():
        # 去除当前行首尾空白后，用正则从头匹配该行内容
        m = row_re.match(line.strip())
        # 如果当前行匹配失败，跳过本行进入下一次循环
        if not m:
            continue
        # 提取正则捕获到的冒号后方全部文本内容
        payload = m.group(1)
        # 判断是否还未读取表头，首次匹配到的行作为表头行
        if header is None:
            # 将表头内容以" | "分割，每个字段去除首尾空白后存入header列表
            header = [x.strip() for x in payload.split(" | ")]
            # 表头处理完成，跳过本次循环剩余逻辑，处理下一行
            continue
        # 将数据行内容以第一个" | "分割一次，分割后的每个片段去除首尾空白
        parts = [x.strip() for x in payload.split(" | ", 1)]
        # 组装字典，key取自表头，value取自分割后内容，无第二部分则填空字符串，添加到结果列表
        out.append({header[0]: parts[0], header[1]: parts[1] if len(parts) > 1 else ""})
    # 函数执行完毕，返回包含解析结果列表的字典
    return {"result": out}
```
